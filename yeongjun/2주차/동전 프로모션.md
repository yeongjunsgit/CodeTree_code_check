# 풀이

/*
최소한의 동전을 사용해야한다.
A타입의 동정은 여러개 지급 가능. B타입은 딱 한번만
동전은 최대 1000개, 모든 동전은 값이 다름이 보장
생각나는 방법은 배낭 문제
A타입은 중복 사용이 가능하므로 여러번 등록 B타입은 1번만 사용
*/

import java.util.*;

public class Main {
    public static int backpack(int n, int m, char[] cType, int[] coin) {
        int[][] dp = new int[n+1][m+1];
        for (int i=0; i<=n; ++i) {
            for (int j=0; j<=m; ++j) {
                dp[i][j] = 1001;
            }
        }

        for (int i=1; i<=n; ++i) {
            // 현재 A코인이라면 시작전에 현재 코인을 추가
            if (cType[i-1] == 'A' && coin[i-1] <= m) {
                dp[i][coin[i-1]] = 1;
            }

            for (int j=1; j<=m; ++j) {
                // 이전 값 올리기
                dp[i][j] = Math.min(dp[i][j], dp[i-1][j]);
                // 현재 A코인이라면 
                if (cType[i-1] == 'A') {
                    if (j + coin[i-1] <= m) {
                        dp[i][j + coin[i-1]] = Math.min(dp[i][j + coin[i-1]], Math.min(dp[i-1][j] + 1, dp[i][j] + 1));
                    } 
                }

                // 현재 B코인이라면 i-1 연산값만 확인
                else {
                    if (j + coin[i-1] <= m && dp[i-1][j] != 1001) {
                        dp[i][j + coin[i-1]] = Math.min(dp[i][j + coin[i-1]], dp[i-1][j] + 1);
                    }
                }
            }

            // 현재 B코인이라면 끝난 후 현재 코인을 추가
            if (cType[i-1] == 'B' && coin[i-1] <= m) {
                dp[i][coin[i-1]] = 1;
            }
        }

        return dp[n][m];

    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();
        char[] cType = new char[n];
        int[] coin = new int[n];
        for (int i = 0; i < n; i++) {
            cType[i] = sc.next().charAt(0);
            coin[i] = sc.nextInt();
        }
        // Please write your code here.
        int answer = backpack(n, m, cType, coin);

        if (answer == 1001) System.out.print(-1);
        else System.out.print(answer);

    }
}
