# 아기 고래의 첫 항해

평균 180분

35% 정답률

총 제출 1,374회

넓은 바다에 갓 태어난 아기 고래 한 마리가 있습니다. 아기 고래는 엄마 고래로부터 독립하여 처음으로 넓은 바다를 탐험하려 합니다.

바다는 *N*×*N* 크기의 격자로 표현됩니다. 격자의 각 칸은 헤엄칠 수 있는 바다(0)이거나, 지나갈 수 없는 암초(1)입니다. 격자에서 *i*행 *j*열의 칸을 (*i*,*j*)로 표현합니다.

아기 고래는 위치 (*r*,*c*)에서 출발하며, 처음에 바라보는 방향은 *d*입니다. 방향은 1이 상, 2가 하, 3이 좌, 4가 우를 의미합니다.

아기 고래의 목표는 헤엄칠 수 있는 모든 바다를 탐험하는 것입니다. 아기 고래는 다음 두 단계를 반복하며 이동합니다.

### 1단계: 인접 탐험

아기 고래는 현재 위치에서 상하좌우로 인접한 칸 중, 아직 방문하지 않은 바다 칸이 있다면 다음의 우선순위대로 한 칸 이동합니다.

1. 현재 바라보는 방향으로 직진
2. 좌회전(90도 반시계 방향 회전) 후 직진
3. 우회전(90도 시계 방향 회전) 후 직진
4. 180도 회전 후 직진

이동하면 바라보는 방향은 이동한 방향으로 갱신됩니다. 이 과정을 인접한 칸에 방문 가능한 바다가 없을 때까지 반복합니다.

### 2단계: 가장 가까운 바다로 이동

인접한 칸 모두에 방문 가능한 바다가 없다면, 아직 방문하지 않은 바다 칸 중 현재 위치에서 가장 가까운 칸을 찾아 이동합니다.

- 거리는 상하좌우로 인접한 칸을 한 칸씩 이동하여 도달하는 데 필요한 최소 이동 횟수입니다. 암초는 지나갈 수 없지만, 이미 방문한 바다는 지나갈 수 있습니다.
- 가장 가까운 칸이 여러 개라면, 행 번호가 가장 작은 칸을 선택합니다. 행 번호도 같다면 열 번호가 가장 작은 칸을 선택합니다.
- 선택한 칸까지 최단 거리로 이동합니다. 매 이동마다 선택한 칸까지의 거리가 1 줄어드는 인접한 칸 중 하나로 이동하며, 그러한 칸이 여러 개라면 좌, 하, 우, 상 순서의 우선순위로 선택합니다.
- 도착 후 바라보는 방향은 마지막 이동 방향으로 갱신됩니다.

# 풀이 코드

```java
import java.util.*;
import java.io.*;

class Loca {
    int x;
    int y;

    Loca(int x, int y) {
        this.x = x;
        this.y = y;
    }

}


public class Main {
    // N = 격자 크기, r, c = 아기 고래 초기 위치, d = 아기 고래 초기 시선 방향
    static int N, r, c, d;
    static int[][] directions = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};
    // 다른 바다 추적 중의 이동 순서 (좌 하 우 상)
    static int[] searchDir = {2, 1, 3, 0};
    static int[] leftDir = {2, 3, 1, 0};
    static int[] rightDir = {3, 2, 0, 1};
    static int[] reverseDir = {1, 0, 3, 2};
    // 현재 위치에서 주변 바다를 이동하는 메서드 / 아기 고래의 이동 순서 (1. 바라보는 방향, 2. 좌회전 후 직진, 3. 우회전 후 직진, 4. 180도(뒤)로 회전 후 직진)
    public static void goAround(int[][] board, int[][] visited, int[][] colors, int aCnt) {
        int nowCnt = 1;
        while (nowCnt < aCnt) {
            // 현재 가는 방향으로 갈 수 있는지 확인
            if (1 <= r + directions[d][0] 
                && r + directions[d][0] <= N 
                && 1 <= c + directions[d][1] 
                && c + directions[d][1] <= N 
                && colors[r + directions[d][0]][c + directions[d][1]] == 1
                && visited[r + directions[d][0]][c + directions[d][1]] != 1) {
                    r += directions[d][0];
                    c += directions[d][1];
            }
            // 좌회전 후 직진으로 갈 수 있는지 확인
            else if (1 <= r + directions[leftDir[d]][0]
                && r + directions[leftDir[d]][0] <= N
                && 1 <= c + directions[leftDir[d]][1]
                && c + directions[leftDir[d]][1] <= N
                && colors[r + directions[leftDir[d]][0]][c + directions[leftDir[d]][1]] == 1
                && visited[r + directions[leftDir[d]][0]][c + directions[leftDir[d]][1]] != 1) {
                    d = leftDir[d];
                    r += directions[d][0];
                    c += directions[d][1];
            }
            // 우회전 후 직진으로 갈 수 있는지 확인
            else if (1 <= r + directions[rightDir[d]][0]
                && r + directions[rightDir[d]][0] <= N
                && 1 <= c + directions[rightDir[d]][1]
                && c + directions[rightDir[d]][1] <= N
                && colors[r + directions[rightDir[d]][0]][c + directions[rightDir[d]][1]] == 1
                && visited[r + directions[rightDir[d]][0]][c + directions[rightDir[d]][1]] != 1) {
                    d = rightDir[d];
                    r += directions[d][0];
                    c += directions[d][1];
            }
            // 뒤로 회전 후 갈 수 있는지 확인
            else if (1 <= r + directions[reverseDir[d]][0]
                && r + directions[reverseDir[d]][0] <= N
                && 1 <= c + directions[reverseDir[d]][1]
                && c + directions[reverseDir[d]][1] <= N
                && colors[r + directions[reverseDir[d]][0]][c + directions[reverseDir[d]][1]] == 1
                && visited[r + directions[reverseDir[d]][0]][c + directions[reverseDir[d]][1]] != 1) {
                    d = reverseDir[d];
                    r += directions[d][0];
                    c += directions[d][1];
            }
            // 전부 안되면 가장 가까운 가능한 바다로 이동
            else {
                findNearestOcean(board, visited, colors);
            }
            System.out.println(r + " " + c);
            visited[r][c] = 1;
            nowCnt++;

        }

        return;

    }


    // 다른 바다 추적 중의 이동 순서 (좌 하 우 상)
    public static void findNearestOcean(int[][] board, int[][] visited, int[][] colors) {
        Queue<Loca> tmpQue = new ArrayDeque<>();
        tmpQue.offer(new Loca(r, c));
        int[][] findVisited = new int[N+1][N+1];
        findVisited[r][c] = 1;
        // 향할 곳이 정해졌다면, 우선순위에 맞춰 이동하며 d를 구한다.
        // 매턴마다 bestLoca가 변화하는지 추적하고, 변화했다면 해당 위치가 가장 짧은 위치 중 우선순위가 가장 좋은 위치임으로 해당 위치로 이동
        while (!tmpQue.isEmpty()) {
            Loca bestLoca = new Loca(999, 999);
            int bestDir = 4;
            Queue<Loca> nextQue = new ArrayDeque<>();
            while (!tmpQue.isEmpty()) {
                Loca now = tmpQue.poll();
                for (int i=0; i<4; ++i) {
                    int nx = now.x + directions[searchDir[i]][0];
                    int ny = now.y + directions[searchDir[i]][1];
                    if (1 <= nx && nx <= N && 1 <= ny && ny <= N && board[nx][ny] != 1 && findVisited[nx][ny] == 0) {
                        // 이동 중에 방문한적 없는 바다를 만나면 현재 bestLoca와 비교하여 우선순위가 더 높은 쪽을 기록
                        if (visited[nx][ny] == 0) {
                            if (bestLoca.x > nx || (bestLoca.x == nx && bestLoca.y > ny)) {
                                bestLoca.x = nx;
                                bestLoca.y = ny;
                                bestDir = searchDir[i];
                            }
                        }
                        nextQue.offer(new Loca(nx, ny));
                        findVisited[nx][ny] = 1;

                    }
                } 
            }
            tmpQue = nextQue;
            // 이번 턴에 방문한 적 없는 바다에 도달 했었다면 가장 가까운 바다임으로, 우선순위 체크가 완료된 bestLoca를 현재 위치로 등록
            if (bestLoca.x != 999 && bestLoca.y != 999) {
                r = bestLoca.x;
                c = bestLoca.y;
                d = bestDir;
                return;
            }

        }

    }

    // 먼저 도달 불가능한 바다들을 추려낸다 (산호로 둘러쌓여있어서 바다지만 못가는 곳)
    public static int checkImpossible(int[][] board, int[][] colors) {
        // r, c를 기준으로 bfs를 돈다. 이때, 산호와 격자의 경계로 이루어진 구역 별로 다른 숫자를 마킹한다. 고래가 있는 구역만 유효한 바다가 되는 것이다.
        int cnt = 1;
        colors[r][c] = 1;
        Queue<Loca> que = new ArrayDeque<>();
        que.offer(new Loca(r, c));
        while (!que.isEmpty()) {
            Loca now = que.poll();
            for (int i=0; i<4; ++i) {
                int nx = now.x + directions[i][0];
                int ny = now.y + directions[i][1];
                if (1 <= nx && nx <= N && 1 <= ny && ny <= N && board[nx][ny] != 1 && colors[nx][ny] == 0) {
                    que.offer(new Loca(nx, ny));
                    colors[nx][ny] = 1;
                    cnt++;
                }
            }
        }

        return cnt;

    }



    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        N = Integer.parseInt(st.nextToken());
        r = Integer.parseInt(st.nextToken());
        c = Integer.parseInt(st.nextToken());
        d = Integer.parseInt(st.nextToken());
        d -= 1;
        int[][] board = new int[N + 1][N + 1], visited = new int[N + 1][N + 1], colors = new int[N + 1][N + 1];
        
        for (int i=1; i<=N; ++i) {
            StringTokenizer tst = new StringTokenizer(br.readLine());
            for (int j=1; j<=N; ++j) {
                board[i][j] = Integer.parseInt(tst.nextToken());
            }
        }

        // 시작 위치를 출력
        System.out.println(r + " " + c);
        visited[r][c] = 1;
        // colors를 체크한다;
        int aCnt = checkImpossible(board, colors);
        // 시작지점부터 순회
        goAround(board, visited, colors, aCnt);

    }
}
```
